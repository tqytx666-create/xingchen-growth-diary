# 可选:GitHub Actions 自动部署(需要带 workflow scope 的 token 才能推送到 .github/workflows/)
# 当前用的是手动方式:本地 `npm run build` 后把 dist 推到 gh-pages 分支(见 deploy.sh)。
# 若以后换上有 workflow 权限的 token,把本文件移到 .github/workflows/deploy.yml 即可启用 CI 自动部署。
name: Deploy to GitHub Pages
on:
  push: { branches: [main] }
  workflow_dispatch:
permissions: { contents: read, pages: write, id-token: write }
concurrency: { group: pages, cancel-in-progress: true }
jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: 20 }
      - run: npm ci
      - run: npm run build
      - uses: actions/configure-pages@v5
        with: { enablement: true }
      - uses: actions/upload-pages-artifact@v3
        with: { path: dist }
  deploy:
    needs: build
    runs-on: ubuntu-latest
    environment: { name: github-pages, url: '${{ steps.deployment.outputs.page_url }}' }
    steps:
      - id: deployment
        uses: actions/deploy-pages@v4
